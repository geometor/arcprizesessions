    rows = len(grid)
    cols = len(grid[0])
    visited = [[False for _ in range(cols)] for _ in range(rows)]
    objects = []

    for r in range(rows):

                object_pixels = []

def transform_grid(input_grid):

    cols = len(input_grid[0])
